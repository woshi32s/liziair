function bakeText(text) {
  const t = (text || '').trim();
  if (!t) { GT.ready = false; return; }
  const fsPx = 190;
  const font = '900 ' + fsPx + 'px ' + FAMILY;
  gctx.font = font;
  const mt = gctx.measureText(t);
  const pad = Math.round(fsPx * 0.34);
  const w = Math.max(32, Math.ceil(mt.width + pad * 6));
  const h = Math.max(32, Math.ceil(fsPx * 2.45));
  gcv.width = w; gcv.height = h;
  gctx.font = font;
  gctx.textBaseline = 'middle';
  gctx.textAlign = 'left';
  gctx.fillStyle = '#000';
  gctx.fillRect(0, 0, w, h);
  /* 三个通道各管一件事，全用 lighter 往上叠，互不覆盖：
     G = 高度场（亮 = 高）—— 埂脚那圈浆最低，淌下去的泪痕中等，蜡埂最高。
     R = 埂顶被灯压出来的那道亮脊。   B = 泪珠顶上的一点高光。 */
  gctx.globalCompositeOperation = 'lighter';
  gctx.globalAlpha = 1;
  gctx.filter = 'none';
  gctx.lineJoin = 'round';
  gctx.lineCap = 'round';
  const by = Math.round(h * 0.30);
  const x0 = pad;
  const inkBot = by + Math.min(fsPx * 0.47, (mt.actualBoundingBoxDescent || fsPx * 0.44));

  /* 1) 蜡被字挡住、在埂脚挤出来的一圈浆 */
  gctx.strokeStyle = 'rgba(0,255,0,0.13)';
  gctx.lineWidth = fsPx * 0.40;
  gctx.strokeText(t, x0, by);

  /* 2) 蜡埂本体：一道被堆起来的厚埂 */
  gctx.strokeStyle = 'rgba(0,255,0,0.92)';
  gctx.lineWidth = fsPx * 0.126;
  gctx.strokeText(t, x0, by);
  gctx.fillStyle = 'rgba(0,255,0,0.92)';
  gctx.fillText(t, x0, by);

  /* 3) 泪痕：先有的痕迹。蜡从埂里溢出来、顺着坡往下淌、走到半路
        就停在原地凝住，末端积成一颗圆珠。字只是把蜡挡住的那一段。 */
  const nT = Math.max(8, Math.round(mt.width / (fsPx * 0.088)));
  let sd2 = 2166136261;
  for (let k = 0; k < t.length; k++) sd2 = ((sd2 ^ t.charCodeAt(k)) * 16777619) >>> 0;
  const rnd2 = () => { sd2 = (sd2 * 1103515245 + 12345) >>> 0; return sd2 / 4294967296; };
  for (let i = 0; i < nT; i++) {
    const fx = x0 + (i + 0.15 + rnd2() * 0.70) / nT * mt.width;
    const rr = rnd2();
    const len = fsPx * (0.045 + 0.62 * Math.pow(rr, 2.3));
    const wd = fsPx * (0.009 + 0.017 * rnd2());
    const top = inkBot - fsPx * (0.10 + 0.20 * rnd2());
    const ex = fx + (rnd2() - 0.5) * wd * 2.2;
    const ey = inkBot + len;
    gctx.strokeStyle = 'rgba(0,255,0,0.40)';
    gctx.lineWidth = wd;
    gctx.beginPath();
    gctx.moveTo(fx, top);
    gctx.bezierCurveTo(fx + (rnd2() - 0.5) * wd * 1.4, top + (ey - top) * 0.36,
                       ex + (rnd2() - 0.5) * wd * 1.8, top + (ey - top) * 0.74,
                       ex, ey);
    gctx.stroke();
    const br = wd * (0.9 + 0.55 * rnd2());
    gctx.fillStyle = 'rgba(0,255,0,0.34)';
    gctx.beginPath(); gctx.arc(ex, ey, br, 0, 6.2832); gctx.fill();
    gctx.fillStyle = 'rgba(0,0,255,0.75)';
    gctx.beginPath(); gctx.arc(ex - br * 0.26, ey - br * 0.30, br * 0.40, 0, 6.2832); gctx.fill();
  }

  /* 4) 埂顶的亮脊 */
  gctx.strokeStyle = 'rgba(255,0,0,0.85)';
  gctx.lineWidth = fsPx * 0.028;
  gctx.strokeText(t, x0, by);
  gctx.globalCompositeOperation = 'source-over';

  const arr = Array.from(t);
  const chars = [];
  let acc = 0;
  for (const ch of arr) {
    const cw = gctx.measureText(ch).width;
    chars.push({ c: ch, x0: (pad + acc) / w, x1: (pad + acc + cw) / w });
    acc += cw;
  }
  GT.chars = chars;
  GT.w = w; GT.h = h;
  const boxH = BOX_W * (h / w);
  GT.box = [-BOX_W / 2, -boxH / 2, BOX_W, boxH];
  GT.ready = true;
  if (glOk) uploadGlyph();
}
