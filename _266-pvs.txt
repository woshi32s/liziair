void main(){
  float s0 = aSeed.w;
  float bx = uBox.x + uCut * uBox.z;
  float t = uT;
  vec3 p;
  vec3 col;
  float sz, al;

  if (s0 > 0.87) {
    /* 挂在埂沿上、正往下淌的一颗蜡珠：越走越慢，走一小段就凝在那 */
    float lane = aSeed.x;
    float ph = fract(t * 0.045 + aSeed.y);
    float run = ph * ph * uBox.z * 0.085;
    p = vec3(bx + (lane - 0.5) * uBox.z * 0.17,
             uBox.y + uBox.w * 0.47 - run,
             0.004 + 0.009 * aSeed.z);
    col = vec3(1.0, 0.78, 0.52);
    sz = 0.0024 + 0.0038 * aSeed.z;
    al = 0.44 * (1.0 - 0.55 * ph);
  } else if (s0 > 0.36) {
    /* 蜡雾：整片斜面上顺坡慢慢飘过去的一层细雾 */
    float ph = fract(t * 0.014 + aSeed.y);
    p = vec3((aSeed.x - 0.5) * uBox.z * 1.9,
             uBox.y - uBox.w * (0.10 + 1.9 * ph),
             0.018 + 0.20 * aSeed.z * aSeed.z);
    p.x += 0.012 * sin(t * 0.27 + aSeed.y * 11.0);
    col = vec3(1.0, 0.84, 0.62);
    sz = 0.0013 + 0.0024 * fract(aSeed.y * 17.0);
    al = 0.085 * (1.0 - abs(ph * 2.0 - 1.0));
  } else {
    /* 台面上的碎屑 / 砂眼：几乎不动，只是被灯一扫一闪 */
    float tw = 0.5 + 0.5 * sin(t * 1.6 + aSeed.x * 23.0);
    p = vec3((aSeed.x * 2.0 - 1.0) * uBox.z * 1.1,
             uBox.y - uBox.w * (aSeed.y * 2.2),
             0.0015 + 0.0045 * aSeed.z);
    col = vec3(1.0, 0.76, 0.46);
    sz = 0.0009 + 0.0019 * aSeed.z;
    al = 0.05 + 0.20 * tw;
  }
  p.x += 0.004 * sin(t * 0.61 + aSeed.w * 12.0);
  vec4 mv = uView * vec4(p, 1.0);
  mv.xyz += (uRight * (aQ.x - 0.5) + uUp * (aQ.y - 0.5)) * sz;
  vD = -mv.z;
  gl_Position = uProj * mv;
  vQ = aQ;
  vA = al * (0.85 + 0.5 * uBeat + 0.25 * uTre) * smoothstep(0.02, 0.13, vD) * (1.0 - smoothstep(0.9, 2.2, vD));
  vCol = col;
}`;/* 粒子：淌下来的蜡珠 · 顺坡飘的蜡雾 · 台面上的碎屑 */
